อ่านโค้ดให้เห็นเส้นทางข้อมูล

  1. make_* สร้างข้อมูลสาธิต → train_test_split กัน final test
  2. Pipeline เก็บขั้น poly/scale/model → GridSearchCV fit เฉพาะ development
  3. cross-validation เลือกด้วย metric ที่กำหนด → refit ตัวชนะบน development โดยค่าเริ่มต้น
  4. predict test หนึ่งครั้ง → รายงาน metrics; classification ตรวจ classes_ ก่อนหยิบ probability คอลัมน์ positive
  5. Clustering ตัวอย่างเป็นการอธิบาย dataset ทั้งชุด ใช้ internal metrics ไม่อ้างเป็นคะแนนทำนายอนาคต

"""Runnable teaching examples. Synthetic data only; no downloads."""
import numpy as np
import pandas as pd
from sklearn.datasets import make_regression, make_classification, make_blobs
from sklearn.model_selection import train_test_split, GridSearchCV, KFold, StratifiedKFold
from sklearn.pipeline import Pipeline, make_pipeline
from sklearn.preprocessing import StandardScaler, PolynomialFeatures
from sklearn.linear_model import Ridge, LogisticRegression
from sklearn.tree import DecisionTreeClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.cluster import KMeans
from sklearn.metrics import (mean_absolute_error, mean_squared_error, r2_score,
    classification_report, confusion_matrix, roc_auc_score, silhouette_score)

def regression():
    X, y = make_regression(n_samples=240, n_features=3, noise=15, random_state=42)
    X_dev, X_test, y_dev, y_test = train_test_split(X, y, test_size=.2, random_state=42)
    pipe = Pipeline([('poly', PolynomialFeatures(include_bias=False)),
                     ('scale', StandardScaler()), ('model', Ridge())])
    cv = KFold(n_splits=5, shuffle=True, random_state=42)
    search = GridSearchCV(pipe, {'poly__degree': [1, 2], 'model__alpha': [.1, 1, 10]},
                          cv=cv, scoring='neg_mean_squared_error')
    search.fit(X_dev, y_dev)
    # GridSearchCV refits the chosen pipeline on all development data by default.
    pred = search.predict(X_test)
    print('REGRESSION best:', search.best_params_)
    print('Final test:', {'MAE': mean_absolute_error(y_test, pred),
          'RMSE': np.sqrt(mean_squared_error(y_test, pred)), 'R2': r2_score(y_test, pred)})

def classification():
    X, y = make_classification(n_samples=600, n_features=8, n_informative=5,
                               weights=[.8, .2], random_state=42)
    X_dev, X_test, y_dev, y_test = train_test_split(X, y, test_size=.2,
                                                   stratify=y, random_state=42)
    pipe = Pipeline([('scale', StandardScaler()), ('model', LogisticRegression(max_iter=2000))])
    # Compare all candidates on the SAME folds; use macro F1 for this exercise.
    candidates = [
        {'model': [LogisticRegression(max_iter=2000)], 'model__C': [.1, 1, 10]},
        {'model': [KNeighborsClassifier()], 'model__n_neighbors': [3, 7, 15]},
        {'model': [DecisionTreeClassifier(random_state=42)], 'model__max_depth': [3, 6]},
        {'model': [RandomForestClassifier(n_estimators=100, random_state=42)],
         'model__max_depth': [4, None]},
        {'model': [GradientBoostingClassifier(random_state=42)],
         'model__n_estimators': [50, 100], 'model__learning_rate': [.05]},
    ]
    cv = StratifiedKFold(n_splits=4, shuffle=True, random_state=42)
    search = GridSearchCV(pipe, candidates, cv=cv, scoring='f1_macro')
    search.fit(X_dev, y_dev)
    pred = search.predict(X_test)
    positive_index = list(search.classes_).index(1)
    probability = search.predict_proba(X_test)[:, positive_index]
    print('CLASSIFICATION best:', search.best_params_)
    print(confusion_matrix(y_test, pred, labels=[0, 1]))
    print(classification_report(y_test, pred, zero_division=0))
    print('ROC-AUC:', roc_auc_score(y_test, probability))

def clustering():
    X, _ = make_blobs(n_samples=240, centers=3, cluster_std=.8, random_state=42)
    # This is descriptive clustering of this dataset, not a future-data score.
    scaler = StandardScaler()
    Z = scaler.fit_transform(X)
    rows, models = [], {}
    for k in range(2, 7):
        model = KMeans(n_clusters=k, n_init=10, random_state=42).fit(Z)
        rows.append({'k': k, 'inertia': model.inertia_,
                     'silhouette': silhouette_score(Z, model.labels_)})
        models[k] = model
    results = pd.DataFrame(rows)
    chosen_k = int(results.loc[results.silhouette.idxmax(), 'k'])
    model = models[chosen_k]
    print('CLUSTERING descriptive metrics:\n', results.to_string(index=False))
    print('Suggested K (check domain usefulness too):', chosen_k)
    print('Centers in original units:\n', scaler.inverse_transform(model.cluster_centers_))
    profile = pd.DataFrame(X, columns=['x1', 'x2']).assign(cluster=model.labels_)
    print(profile.groupby('cluster').agg(['size', 'mean', 'median']))

if __name__ == '__main__':
    regression()
    classification()
    clustering()

"""Additional runnable chapter 7/12 patterns: mixed columns and learning curves."""
import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.pipeline import Pipeline
from sklearn.linear_model import Ridge
from sklearn.model_selection import (train_test_split, KFold, RandomizedSearchCV,
                                      learning_curve, TimeSeriesSplit, GroupKFold)
from sklearn.metrics import mean_squared_error
from scipy.stats import loguniform

rng = np.random.default_rng(42)
X = pd.DataFrame({'temperature': rng.normal(28, 4, 180),
                  'day_type': rng.choice(['workday', 'holiday'], 180)})
y = 20 + 2 * X.temperature.to_numpy() + 8 * (X.day_type == 'workday') + rng.normal(0, 2, 180)
X.loc[::17, 'temperature'] = np.nan
X_dev, X_test, y_dev, y_test = train_test_split(X, y, test_size=.2, random_state=42)

numeric = Pipeline([('impute', SimpleImputer(strategy='median')),
                    ('scale', StandardScaler())])
categorical = Pipeline([('impute', SimpleImputer(strategy='most_frequent')),
                        ('encode', OneHotEncoder(handle_unknown='ignore'))])
preprocess = ColumnTransformer([('numeric', numeric, ['temperature']),
                                ('categorical', categorical, ['day_type'])])
pipe = Pipeline([('prepare', preprocess), ('model', Ridge())])
cv = KFold(n_splits=4, shuffle=True, random_state=42)
search = RandomizedSearchCV(pipe, {'model__alpha': loguniform(.01, 100)},
                            n_iter=8, cv=cv, scoring='neg_mean_squared_error', random_state=42)
search.fit(X_dev, y_dev)
print('Best alpha:', search.best_params_)

# Diagnostic curves remain on development, never on final-test data.
sizes, train_scores, val_scores = learning_curve(
    search.best_estimator_, X_dev, y_dev, cv=cv,
    train_sizes=[.3, .6, 1.], scoring='neg_mean_squared_error')
print(pd.DataFrame({'training_rows': sizes,
                    'mean_train_RMSE': np.sqrt(-train_scores).mean(axis=1),
                    'mean_val_RMSE': np.sqrt(-val_scores).mean(axis=1)}))
print('Final test RMSE:', np.sqrt(mean_squared_error(y_test, search.predict(X_test))))

# Split examples only: the synthetic regression above is not a time series.
ordered_X = np.arange(60).reshape(-1, 1)
for train_index, val_index in TimeSeriesSplit(n_splits=3, gap=2).split(ordered_X):
    assert train_index.max() < val_index.min()
    print('Temporal fold:', (train_index.min(), train_index.max()),
          '->', (val_index.min(), val_index.max()))
groups = np.repeat(np.arange(12), 5)
for train_index, val_index in GroupKFold(n_splits=3).split(ordered_X, groups=groups):
    assert not set(groups[train_index]) & set(groups[val_index])
print('Group folds: no person/group appears on both sides.')

แผนที่ protocol ของแล็บ

งานข้อมูลและการเลือกจุดที่ต้องรักษา
P01development 90 วัน: train 72 แรก, validation 18 ท้าย; Simple=temp, Multiple=temp+working_dayรายงาน MAE/RMSE kWh และ R²; ห้ามใช้ final test เลือก
P02sin(x)+Normal(0,.25), x∼Uniform(0,6); train35 seeds11/22/33; validation200 seed2026; degree1/3/12ใช้ validation เดิมทั้ง 9 การทดลอง เก็บ prediction ที่ x=2; 3 seeds ยังไม่ใช่ข้อพิสูจน์ bias–variance ครบถ้วน
P03A=temp+day; B=A+temp²; C/D=scale(B)→Ridge(.1/10)เลือก validation RMSE; ต่าง <.01 kWh ให้แบบง่ายกว่าแล้ว alpha ต่ำกว่า; refit development90 จากนั้น test30 ครั้งสุดท้าย
P04model A บน development90; ordinary pairs และ moving-block pairs block7; 300 รอบ/วิธี seed20260922สุ่ม X,y คู่กัน; block starts 0..83; percentile2.5/97.5 ของ coefficient temp; ไม่ใช่ prediction interval
P05train50 seed41, val100 seed42, test100 seed43; x∼Uniform(0,10), noise SD=.15N12, width .8; OLS/Ridge(.1)/Lasso(.001); tie≤1e−6 เลือก OLS→Ridge→Lasso; ไม่ refit ก่อน test

"""Reference practice solutions for P01-P05; preserve original notebooks.
Run from any working directory. Produces results in exam-prep/results.
"""
from pathlib import Path
import json, platform
import numpy as np
import pandas as pd
import sklearn
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression, Ridge, Lasso
from sklearn.preprocessing import StandardScaler, PolynomialFeatures
from sklearn.pipeline import make_pipeline
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from sklearn.base import BaseEstimator, TransformerMixin
from sklearn.utils.validation import check_array, check_is_fitted

BASE = Path(__file__).resolve().parents[2]
OUT = BASE / 'exam-prep' / 'results'
OUT.mkdir(exist_ok=True)

def rmse(y, pred): return float(np.sqrt(mean_squared_error(y, pred)))
def metrics(y, pred):
    return {'MAE': float(mean_absolute_error(y, pred)), 'RMSE': rmse(y, pred),
            'R2': float(r2_score(y, pred))}

def features(df, quadratic=False):
    X = df[['temp_forecast_c', 'working_day']].copy()
    if quadratic: X['temp_squared'] = X.temp_forecast_c ** 2
    return X

def energy_lab():
    dev = pd.read_csv(BASE / 'quiz' / 'regression_development.csv', parse_dates=['date'])
    assert dev.date.is_monotonic_increasing and len(dev) == 90
    train, val = dev.iloc[:72], dev.iloc[72:]
    p01 = []
    for name, cols in [('Simple', ['temp_forecast_c']),
                       ('Multiple', ['temp_forecast_c', 'working_day'])]:
        model = LinearRegression().fit(train[cols], train.energy_kwh)
        p01.append({'model': name, **metrics(val.energy_kwh, model.predict(val[cols]))})
    pd.DataFrame(p01).to_csv(OUT / 'p01.csv', index=False)
    specs = [('A', False, LinearRegression()), ('B', True, LinearRegression()),
             ('C', True, make_pipeline(StandardScaler(), Ridge(alpha=.1))),
             ('D', True, make_pipeline(StandardScaler(), Ridge(alpha=10)))]
    rows = []
    for name, quadratic, model in specs:
        model.fit(features(train, quadratic), train.energy_kwh)
        rows.append({'model': name, **metrics(val.energy_kwh, model.predict(features(val, quadratic)))})
    # Protocol: within <0.01 kWh of the best, prefer simpler; then lower alpha.
    best = min(r['RMSE'] for r in rows)
    chosen = next(r['model'] for r in rows if r['RMSE'] - best < .01)
    pd.DataFrame(rows).to_csv(OUT / 'p03_validation.csv', index=False)
    pred_a = specs[0][2].predict(features(val))
    residual = val.energy_kwh.to_numpy() - pred_a
    fig, axes = plt.subplots(1, 2, figsize=(10, 4))
    axes[0].scatter(pred_a, residual)
    axes[0].set(xlabel='Fitted energy (kWh)', ylabel='Residual = actual - predicted', title='Validation residuals: model A')
    axes[1].plot(val.date, residual, marker='o')
    axes[1].set(xlabel='Date', ylabel='Residual (kWh)', title='Residuals over time')
    for ax in axes: ax.axhline(0, color='black', linewidth=.8)
    fig.autofmt_xdate(); fig.tight_layout(); fig.savefig(OUT / 'p03_residuals.png', dpi=140); plt.close(fig)
    # Only now open final-test data, refit chosen model, and report once.
    name, quadratic, model = next(s for s in specs if s[0] == chosen)
    model.fit(features(dev, quadratic), dev.energy_kwh)
    test = pd.read_csv(BASE / 'quiz' / 'regression_final_test.csv', parse_dates=['date'])
    assert len(test) == 30 and dev.date.max() < test.date.min()
    final = {'chosen': chosen, **metrics(test.energy_kwh, model.predict(features(test, quadratic)))}
    (OUT / 'p03_final.json').write_text(json.dumps(final, indent=2), encoding='utf-8')
    # P04: each method starts its own generator with the same specified seed.
    boot = []
    X, y, n = features(dev), dev.energy_kwh, len(dev)
    for method in ['ordinary', 'moving_block']:
        rng = np.random.default_rng(20260922)
        values = []
        for _ in range(300):
            if method == 'ordinary': indices = rng.integers(0, n, size=n)
            else:
                starts = rng.integers(0, n - 7 + 1, size=int(np.ceil(n / 7)))
                indices = np.concatenate([np.arange(s, s + 7) for s in starts])[:n]
            fit = LinearRegression().fit(X.iloc[indices], y.iloc[indices])
            values.append(fit.coef_[0])
        lo, hi = np.percentile(values, [2.5, 97.5])
        boot.append({'method': method, 'lower_2.5': lo, 'upper_97.5': hi})
    pd.DataFrame(boot).to_csv(OUT / 'p04.csv', index=False)
    print('P01', p01, '\nP03 validation', rows, '\nP03 final', final, '\nP04', boot)

def bias_variance_lab():
    def sine(seed, n):
        rng = np.random.default_rng(seed)
        x = rng.uniform(0, 6, (n, 1))
        return x, np.sin(x[:, 0]) + rng.normal(0, .25, n)
    xv, yv = sine(2026, 200)
    rows = []
    for seed in [11, 22, 33]:
        x, y = sine(seed, 35)
        for degree in [1, 3, 12]:
            model = make_pipeline(PolynomialFeatures(degree=degree, include_bias=False),
                                  StandardScaler(), LinearRegression()).fit(x, y)
            rows.append({'seed': seed, 'degree': degree, 'train_RMSE': rmse(y, model.predict(x)),
                         'val_RMSE': rmse(yv, model.predict(xv)),
                         'prediction_at_2': float(model.predict([[2.]])[0])})
    pd.DataFrame(rows).to_csv(OUT / 'p02.csv', index=False)
    print('P02\n', pd.DataFrame(rows).to_string(index=False))

class GaussianFeatures(TransformerMixin, BaseEstimator):
    def __init__(self, N=12, width_factor=.8):
        self.N = N
        self.width_factor = width_factor
    def fit(self, X, y=None):
        X = check_array(X)
        if X.shape[1] != 1 or not isinstance(self.N, (int, np.integer)) or self.N < 2:
            raise ValueError('One feature and integer N >= 2 required')
        if not np.isfinite(self.width_factor) or self.width_factor <= 0 or X.max() <= X.min():
            raise ValueError('Positive finite width and nonconstant training X required')
        self.n_features_in_ = 1
        self.centers_ = np.linspace(X.min(), X.max(), self.N)
        self.sigma_ = self.width_factor * (self.centers_[1] - self.centers_[0])
        return self
    def transform(self, X):
        check_is_fitted(self, ['centers_', 'sigma_'])
        X = check_array(X)
        if X.shape[1] != 1: raise ValueError('One feature required')
        return np.exp(-.5 * ((X[:, 0, None] - self.centers_[None, :]) / self.sigma_) ** 2)

def gaussian_lab():
    def data(seed, n):
        rng = np.random.default_rng(seed)
        X = rng.uniform(0, 10, (n, 1))
        return X, np.sin(X[:, 0]) + rng.normal(0, .15, n)
    x, y = data(41, 50); xv, yv = data(42, 100)
    poly = PolynomialFeatures(degree=3, include_bias=False).fit_transform(x[:3])
    basis = GaussianFeatures(N=12, width_factor=.8)
    B = basis.fit_transform(x)
    scaler = StandardScaler()
    Z = scaler.fit_transform(B); Zv = scaler.transform(basis.transform(xv))
    models = [('OLS', LinearRegression()), ('Ridge', Ridge(alpha=.1)),
              ('Lasso', Lasso(alpha=.001, max_iter=100000, tol=1e-6))]
    rows = []
    for name, model in models:
        model.fit(Z, y)
        rows.append({'model': name, 'train_RMSE': rmse(y, model.predict(Z)),
                     'val_RMSE': rmse(yv, model.predict(Zv)),
                     'zero_coefficients': int(np.sum(model.coef_ == 0))})
    best = min(r['val_RMSE'] for r in rows)
    index = next(i for i, r in enumerate(rows) if r['val_RMSE'] - best <= 1e-6)
    name, model = models[index]
    # P05 explicitly forbids refitting; retain train-fitted model, centers and scale.
    xt, yt = data(43, 100)
    final = {'chosen': name, 'test_RMSE': rmse(yt, model.predict(scaler.transform(basis.transform(xt)))),
             'centers': basis.centers_.tolist(), 'sigma': float(basis.sigma_),
             'train_basis_shape': list(B.shape), 'polynomial_first_three': poly.tolist()}
    pd.DataFrame(rows).to_csv(OUT / 'p05_validation.csv', index=False)
    (OUT / 'p05_final.json').write_text(json.dumps(final, indent=2), encoding='utf-8')
    grid = np.linspace(0, 10, 300)[:, None]
    fig, axes = plt.subplots(1, 2, figsize=(10, 4))
    axes[0].plot(grid[:, 0], basis.transform(grid)); axes[0].set(title='12 train-fitted Gaussian bases', xlabel='x')
    axes[1].scatter(x[:, 0], y, s=15, alpha=.5, label='train')
    axes[1].plot(grid[:, 0], np.sin(grid[:, 0]), label='true mean')
    axes[1].plot(grid[:, 0], model.predict(scaler.transform(basis.transform(grid))), label=name)
    axes[1].set(title='Selected model (no refit)', xlabel='x'); axes[1].legend()
    fig.tight_layout(); fig.savefig(OUT / 'p05_basis.png', dpi=140); plt.close(fig)
    print('P05', rows, '\nP05 final', final)

if __name__ == '__main__':
    versions = {'python': platform.python_version(), 'numpy': np.__version__,
                'pandas': pd.__version__, 'sklearn': sklearn.__version__}
    (OUT / 'versions.json').write_text(json.dumps(versions, indent=2), encoding='utf-8')
    energy_lab(); bias_variance_lab(); gaussian_lab()
